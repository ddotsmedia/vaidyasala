TASK 1: DATABASE MIGRATIONS (10 min)
=========================================================
Create: apps/web/prisma/migrations/[timestamp]_add_dosha_tables/migration.sql

```sql
CREATE TABLE IF NOT EXISTS "DoshaAssessment" (
  "id" TEXT PRIMARY KEY DEFAULT gen_random_uuid(),
  "userId" TEXT UNIQUE,
  "vata" INTEGER DEFAULT 0,
  "pitta" INTEGER DEFAULT 0,
  "kapha" INTEGER DEFAULT 0,
  "dominantDosha" VARCHAR(10),
  "submittedAt" TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  "updatedAt" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS "WellnessEntry" (
  "id" TEXT PRIMARY KEY DEFAULT gen_random_uuid(),
  "userId" TEXT NOT NULL,
  "date" DATE NOT NULL,
  "energy" INTEGER,
  "digestion" INTEGER,
  "mood" INTEGER,
  "sleep" INTEGER,
  "notes" TEXT,
  "createdAt" TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE("userId", "date")
);

CREATE TABLE IF NOT EXISTS "DoshaHistory" (
  "id" TEXT PRIMARY KEY DEFAULT gen_random_uuid(),
  "userId" TEXT NOT NULL,
  "doshaDate" DATE NOT NULL,
  "season" VARCHAR(20),
  "vata" INTEGER,
  "pitta" INTEGER,
  "kapha" INTEGER,
  "dominantDosha" VARCHAR(10),
  UNIQUE("userId", "doshaDate")
);

CREATE INDEX IF NOT EXISTS "idx_dosha_user" ON "DoshaAssessment"("userId");
CREATE INDEX IF NOT EXISTS "idx_wellness_user_date" ON "WellnessEntry"("userId", "date");
```

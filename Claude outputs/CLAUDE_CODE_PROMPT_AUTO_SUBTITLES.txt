```ts
import { Deepgram } from "@deepgram/sdk";

const deepgram = new Deepgram(process.env.DEEPGRAM_API_KEY);

export async function transcribeYoutubeVideo(youtubeId: string) {
  const videoUrl = `https://www.youtube.com/watch?v=${youtubeId}`;

  const response = await deepgram.listen.preRecorded.transcribeUrl(
    { url: videoUrl },
    {
      model: "nova-3",
      language: "en",
      smart_format: true,
      diarize: true,
      punctuate: true,
      paragraphs: true,
    }
  );

  return {
    transcript: response.results.channels[0].alternatives[0].transcript,
    confidence: response.results.channels[0].alternatives[0].confidence,
    words: response.results.channels[0].alternatives[0].words,
    paragraphs: response.results.channels[0].alternatives[0].paragraphs,
  };
}

export function generateVTT(words: any[]) {
  let vtt = "WEBVTT\n\n";

  words.forEach((word, i) => {
    const start = formatTime(word.start);
    const end = formatTime(word.end);
    vtt += `${start} --> ${end}\n${word.punctuated_word}\n\n`;
  });

  return vtt;
}

function formatTime(seconds: number) {
  const h = Math.floor(seconds / 3600);
  const m = Math.floor((seconds % 3600) / 60);
  const s = Math.floor(seconds % 60);
  const ms = Math.floor((seconds % 1) * 1000);
  return `${h.toString().padStart(2, "0")}:${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}.${ms.toString().padStart(3, "0")}`;
}
```

TASK 5: SCRIPT TO PROCESS ALL 503 VIDEOS (30 min)
=========================================================
Create: apps/web/scripts/process-all-subtitles.ts
```ts
import { PrismaClient } from "@prisma/client";
import { transcribeYoutubeVideo, generateVTT } from "../lib/deepgram";

const prisma = new PrismaClient();

async function processAllSubtitles() {
  const videos = await prisma.video.findMany({
    where: { status: "published" },
  });

  console.log(`Processing ${videos.length} videos...`);

  for (const video of videos) {
    try {
      console.log(`[${videos.indexOf(video) + 1}/${videos.length}] Processing ${video.titleMl}...`);

      const transcription = await transcribeYoutubeVideo(video.youtubeId);
      const vtt = generateVTT(transcription.words);

      await prisma.subtitle.upsert({
        where: { videoId: video.id },
        update: {
          vtt,
          json: { transcript: transcription.transcript, words: transcription.words },
          status: "completed",
          transcribedAt: new Date(),
        },
        create: {
          videoId: video.id,
          vtt,
          json: { transcript: transcription.transcript, words: transcription.words },
          status: "completed",
          transcribedAt: new Date(),
        },
      });

      // Rate limit: 1 video per 10 seconds
      await new Promise(r => setTimeout(r, 10000));
    } catch (error) {
      console.error(`❌ Failed: ${video.titleMl}`, error);
    }
  }

  console.log("✅ Done!");
  await prisma.$disconnect();
}

processAllSubtitles();
```

```ts
import { PrismaClient } from "@prisma/client";

const prisma = new PrismaClient();

async function verifySubtitles() {
  const videos = await prisma.video.findMany({
    where: { status: "published" },
    include: { subtitle: true }
  });

  const withSubtitles = videos.filter(v => v.subtitle).length;
  const total = videos.length;
  const percent = ((withSubtitles / total) * 100).toFixed(1);

  console.log(`
  📊 SUBTITLE STATS
  ═════════════════
  Total videos: ${total}
  With subtitles: ${withSubtitles}
  Coverage: ${percent}%

  Status breakdown:
  - Completed: ${await prisma.subtitle.count({ where: { status: "completed" } })}
  - Processing: ${await prisma.subtitle.count({ where: { status: "processing" } })}
  - Failed: ${await prisma.subtitle.count({ where: { status: "failed" } })}
  `);

  await prisma.$disconnect();
}

verifySubtitles();
```
